\chapter*{Danh mục từ viết tắt}
\addcontentsline{toc}{chapter}{Danh mục từ viết tắt}
\begin{description}[labelwidth=2.6cm, leftmargin=2.9cm, style=sameline]
  \item[ALM] Adaptive Logic Module (phần tử logic của Intel Cyclone V)
  \item[ASIC] Application-Specific Integrated Circuit
  \item[Avalon-MM] Avalon Memory-Mapped interface (bus của Intel/Altera)
  \item[CBUF] bộ đệm tích lũy INT32 (Accumulator BUFfer) trong \NPU{}
  \item[CNN] Convolutional Neural Network -- mạng nơ-ron tích chập
  \item[CPU] Central Processing Unit
  \item[DMA] Direct Memory Access
  \item[DSP] Digital Signal Processing block (khối nhân cứng trên FPGA)
  \item[FPGA] Field-Programmable Gate Array
  \item[FSM] Finite State Machine
  \item[GEMM] GEneral Matrix Multiplication -- nhân ma trận tổng quát
  \item[HPS] Hard Processor System (cụm ARM trên Cyclone V SoC)
  \item[im2col] image-to-column -- biến đổi tích chập thành nhân ma trận
  \item[INT8/INT32] số nguyên có dấu 8/32 bit
  \item[IS] Input-Stationary dataflow
  \item[ISA] Instruction Set Architecture
  \item[LSTM] Long Short-Term Memory
  \item[LUT] Look-Up Table -- bảng tra (cũng là phần tử logic của FPGA)
  \item[MAC] Multiply-ACcumulate -- nhân-cộng dồn
  \item[MMIO] Memory-Mapped Input/Output
  \item[MNIST] tập dữ liệu chữ số viết tay
  \item[NPU] Neural Processing Unit -- ở đây là bộ tăng tốc của đề tài (\code{npu\_avmm})
  \item[OS] Output-Stationary dataflow
  \item[PE] Processing Element -- phần tử xử lý của mảng systolic
  \item[PTQ] Post-Training Quantization -- lượng tử hóa sau huấn luyện
  \item[QBUF] bộ đệm kết quả INT8 sau hậu xử lý
  \item[ABUF] bộ đệm kích hoạt INT8
  \item[ReLU] Rectified Linear Unit
  \item[RISC-V] kiến trúc tập lệnh mở RISC-V
  \item[RNN] Recurrent Neural Network -- mạng nơ-ron hồi quy
  \item[RTL] Register-Transfer Level
  \item[SoC] System-on-Chip
  \item[TPU] Tensor Processing Unit (Google)
  \item[VCD] Value Change Dump (dạng sóng mô phỏng)
  \item[WS] Weight-Stationary dataflow
\end{description}
